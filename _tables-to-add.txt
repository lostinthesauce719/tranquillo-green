  rateLimits: defineTable({
    key: v.string(),
    count: v.number(),
    resetAt: v.number(),
  }).index("by_key", ["key"]),

  // ─── NOTIFICATION READ STATE ──────────────────────────────────────
  // Per-user read/dismiss state over derived notification sources
  // (compliance alerts, audit events). The feed itself is composed live.
  notificationStates: defineTable({
    userId: v.id("users"),
    sourceId: v.string(), // _id of the underlying alert/event
    readAt: v.optional(v.number()),
    dismissedAt: v.optional(v.number()),
  }).index("by_user", ["userId"]).index("by_user_source", ["userId", "sourceId"]),

  // ─── MARKETING CAMPAIGNS ──────────────────────────────────────────
  campaigns: defineTable({
    companyId: v.id("cannabisCompanies"),
    name: v.string(),
    channel: v.string(), // "google", "meta", "leafly", "weedmaps", "email", "other"
    status: v.union(v.literal("draft"), v.literal("active"), v.literal("paused"), v.literal("completed")),
    budget: v.number(),
    spent: v.number(),
    impressions: v.number(),
    clicks: v.number(),
    leads: v.number(),
    startDate: v.optional(v.string()), // YYYY-MM-DD
    endDate: v.optional(v.string()),
    notes: v.optional(v.string()),
    createdAt: v.number(),
    updatedAt: v.number(),
  }).index("by_company", ["companyId"]),

  // ─── CONTENT ITEMS ────────────────────────────────────────────────
  contentItems: defineTable({
    companyId: v.id("cannabisCompanies"),
    title: v.string(),
    channel: v.string(), // "twitter", "linkedin", "reddit", "email", "blog", "other"
    status: v.union(v.literal("draft"), v.literal("scheduled"), v.literal("published")),
    scheduledFor: v.optional(v.string()), // YYYY-MM-DD
    body: v.optional(v.string()),
    createdAt: v.number(),
    updatedAt: v.number(),
  }).index("by_company", ["companyId"]),

  // ─── CPA CLIENT LINKS ─────────────────────────────────────────────
  // Grants an external CPA (keyed by email) read access to a company's
  // close/compliance summary via the CPA portal.
  cpaClientLinks: defineTable({
    companyId: v.id("cannabisCompanies"),
    cpaEmail: v.string(), // normalized lowercase
    status: v.union(v.literal("active"), v.literal("revoked")),
    addedBy: v.string(), // actor label
    createdAt: v.number(),
  }).index("by_company", ["companyId"]).index("by_email", ["cpaEmail"]),

  // ─── METRC PACKAGE RECONCILIATION ─────────────────────────────────
  metrcPackages: defineTable({
    companyId: v.id("cannabisCompanies"),
    packageLabel: v.string(), // Metrc tag
    product: v.string(),
    uom: v.string(), // "g", "lb", "ea"
    metrcQty: v.number(),
    bookQty: v.number(),
    status: v.union(v.literal("matched"), v.literal("variance"), v.literal("missing_in_books"), v.literal("resolved")),
    resolutionNote: v.optional(v.string()),
    source: v.union(v.literal("manual"), v.literal("api")),
    lastSyncAt: v.number(),
    createdAt: v.number(),
    updatedAt: v.number(),
  }).index("by_company", ["companyId"]).index("by_company_label", ["companyId", "packageLabel"]),

  // ─── USER-COMPANY LINKS (multi-operation) ─────────────────────────
  // Lets one user own/manage multiple operations; users.companyId remains
  // the ACTIVE tenant, switched via users.switchActiveCompany.
  userCompanyLinks: defineTable({
    userId: v.id("users"),
    companyId: v.id("cannabisCompanies"),
    role: v.union(v.literal("owner"), v.literal("controller"), v.literal("accountant"), v.literal("viewer")),
    createdAt: v.number(),
  }).index("by_user", ["userId"]).index("by_user_company", ["userId", "companyId"]),

  // ─── EMPLOYEE ROSTER ──────────────────────────────────────────────
  // Payroll roster with per-employee plant-touching allocation used by
  // the 280E labor allocation workflow.
  employees: defineTable({
    companyId: v.id("cannabisCompanies"),
    name: v.string(),
    role: v.string(),
    payRate: v.number(), // hourly rate
    hoursPerPeriod: v.number(),
    allocation: v.number(), // % plant-touching (0-100)
    active: v.boolean(),
    externalRef: v.optional(v.string()), // Gusto/payroll external ref
    createdAt: v.number(),
    updatedAt: v.number(),
  }).index("by_company", ["companyId"]),

  // Tracks employee time by activity classification for 280E allocation.
  // Production activities = capitalizable (COGS). Non-production = 280E limited.
